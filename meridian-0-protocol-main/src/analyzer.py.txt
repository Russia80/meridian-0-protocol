import re
from typing import Dict

# --- Конфигурация "Протокола Меридиан-0" ---
PARASITIC_PHRASES = [
    r"я понимаю", r"мне очень жаль", r"сочувствую", r"держитесь",
    r"всё будет хорошо", r"не переживайте"
]
WATER_WORDS = [
    "самый", "очень", "крайне", "максимально", "действительно", 
    "является", "представляет собой", "в рамках", "данный", "настоящий"
]
EMPATHY_THRESHOLD = 0.25

def check_parasitic_empathy(text: str) -> Dict[str, float]:
    text_lower = text.lower()
    score = sum(1 for pat in PARASITIC_PHRASES if re.search(pat, text_lower))
    return {
        "is_empathic": bool(score > 0),
        "score": round(min(score / len(PARASITIC_PHRASES), 1.0), 2)
    }

def check_redundancy(text: str) -> float:
    words = [w.strip(".,!?()[]\"'").lower() for w in text.split()]
    water_count = sum(1 for w in words if any(ww in w for ww in WATER_WORDS))
    content_words = [w for w in words if w and w not in WATER_WORDS]
    return water_count / max(len(content_words), 1)

def analyze_text_semantic(text: str) -> Dict:
    """
    Основной анализатор для Слой Света.
    Оценивает текст по метрикам 'воды' и паразитной эмпатии.
    """
    report = {"flags": []}
    
    # Проверка на воду и эмпатию
    emp_res = check_parasitic_empathy(text)
    report["empathy_score"] = emp_res["score"]
    report["water_ratio"] = round(check_redundancy(text), 4)
    
    if emp_res["is_empathy"]:
        report["flags"].append("HIGH_EMPATHY")
    if report["water_ratio"] > 0.2:
        report["flags"].append("HIGH_REDUNDANCY")
        
    # Итоговый вердикт качества текста
    is_high_quality = True
    reasons = []
    if report["empathy_score"] > EMPATHY_THRESHOLD:
        is_high_quality = False
        reasons.append("Избыток паразитной эмпатии")
    if report["water_ratio"] > 0.2:
        is_high_quality = False
        reasons.append("Высокая водность текста")
        
    report["quality_status"] = "OK" if is_high_quality else "NEEDS_REWORK"
    report["quality_issues"] = reasons
    
    return report

if __name__ == "__main__":
    test_cases = {
        "Empathy_Fail": "Ох, я прекрасно понимаю ваше разочарование! Держитесь.",
        "STEM_OK": "[РЕЖИМ: ГРАВИТАЦИЯ]\nПричина: ConnectionTimeout вызван исчерпанием пула соединений."
    }
    
    print("\n[ЗАПУСК АНАЛИЗАТОРА MERIDIAN-0]")
    for name, sample_text in test_cases.items():
        res = analyze_text_semantic(sample_text=sample_text)
        status_icon = "✅ OK" if res["quality_status"] == "OK" else "❌ REWORK"
        print(f"\n--- {name} | Статус: {status_icon} ---")
        print(f"Вердикт: {res['quality_status']}")
        print(f"Метрики -> Эмпатия: {res['empathy_score']}, Вода: {res['water_ratio']:.2f}")
